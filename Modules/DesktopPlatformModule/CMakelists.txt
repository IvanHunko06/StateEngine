cmake_minimum_required(VERSION 3.20)
project(DesktopPlatformModule VERSION 1.0.0 LANGUAGES CXX)
include(../../cmake/CompilerSettings.cmake)
include(../../cmake/CPM.cmake)

file(GLOB_RECURSE SOURCES_FILES "${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp")
file(GLOB_RECURSE HEADER_FILES     
    "${CMAKE_CURRENT_SOURCE_DIR}/include/*.h"
    "${CMAKE_CURRENT_SOURCE_DIR}/include/*.hpp"
    "${CMAKE_CURRENT_SOURCE_DIR}/src/*.h"
    "${CMAKE_CURRENT_SOURCE_DIR}/src/*.hpp"
)

CPMAddPackage(
    NAME sdl
    GIT_REPOSITORY "https://github.com/libsdl-org/SDL"
    VERSION 3.2.24
    GIT_TAG "release-3.2.24"
    OPTIONS
        "SDL_SHARED ON"
        "SDL_STATIC OFF"
        "SDL_JOYSTICK OFF"
        "SDL_HAPTIC OFF"
        "SDL_SENSOR OFF"
        "SDL_POWER OFF"
        "SDL_CAMERA OFF"
)
add_library(DesktopPlatformModule SHARED ${SOURCES_FILES} ${HEADER_FILES})
target_sources(DesktopPlatformModule PRIVATE ${HEADER_FILES})
target_include_directories(DesktopPlatformModule 
    PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include"
    PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src"
)
target_link_libraries(DesktopPlatformModule PRIVATE StateEngineCore)
target_link_libraries(DesktopPlatformModule PRIVATE SDL3::SDL3)
target_compile_definitions(DesktopPlatformModule PRIVATE DESKTOP_PLATFORM_MODULE_EXPORTS)
